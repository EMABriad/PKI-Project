from cryptography.hazmat.primitives.asymmetric import rsa, padding
from cryptography.hazmat.primitives import hashes

print("\n=== PKI Practical Implementation: RSA Encryption, Decryption & Digital Signature ===\n")

private_key = rsa.generate_private_key(
    public_exponent=65537,
    key_size=2048
)

public_key = private_key.public_key()

message = b"This is a PKI secure message"

encrypted_message = public_key.encrypt(
    message,
    padding.OAEP(
        mgf=padding.MGF1(algorithm=hashes.SHA256()),
        algorithm=hashes.SHA256(),
        label=None
    )
)

decrypted_message = private_key.decrypt(
    encrypted_message,
    padding.OAEP(
        mgf=padding.MGF1(algorithm=hashes.SHA256()),
        algorithm=hashes.SHA256(),
        label=None
    )
)

signature = private_key.sign(
    message,
    padding.PSS(
        mgf=padding.MGF1(hashes.SHA256()),
        salt_length=padding.PSS.MAX_LENGTH
    ),
    hashes.SHA256()
)

verification = public_key.verify(
    signature,
    message,
    padding.PSS(
        mgf=padding.MGF1(hashes.SHA256()),
        salt_length=padding.PSS.MAX_LENGTH
    ),
    hashes.SHA256()
)

print("Encrypted Message:", encrypted_message)
print("\nDecrypted Message:", decrypted_message.decode())
print("\nSignature Created Successfully")

print("\nSignature Verification: VALID")



#This practical implementation demonstrates the core principles of Public Key Infrastructure (PKI) using RSA asymmetric cryptography. The experiment includes key generation, encryption and decryption processes, and digital signature creation and verification. The results confirm the confidentiality, integrity, and authentication capabilities provided by PKI.
